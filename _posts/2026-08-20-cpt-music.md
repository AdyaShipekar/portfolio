In JavaScript:

{% capture music_js_code %}
console.log("Music Recommendation System");

let songs = ["Happier by Bastille", "Focused Lofi Beats", "Last Man Standing by Livingston", "Shake It Off by Taylor Swift"];

let effects = [
    "Improve mood",
    "Focus",
    "Calm down",
    "Become energized"
];

let lyrics = ["yes", "no", "no", "yes"];

let desiredEffect = prompt("What do you want to do?");
let avoidLyrics = prompt("Avoid lyrics? (yes/no)");

let index = 0;
let found = false;

for (let effect of effects) {
    if (effect === desiredEffect) {
        if (avoidLyrics === "no" || lyrics[index] === "no") {
            console.log("Recommended song: " + songs[index]);
            found = true;
        }
    }

    index = index + 1;
}

if (found === false) {
    console.log("No matching song found");
}
{% endcapture %}

(function () {
  /* ── Python: Pyodide ── */
  var _pyReady = null;
  function getPyodide() {
    if (!_pyReady) {
      _pyReady = new Promise(function (resolve, reject) {
        var s = document.createElement('script');
        s.src = 'https://cdn.jsdelivr.net/pyodide/v0.27.0/full/pyodide.js';
        s.onload = function () {
          loadPyodide().then(function (py) {
            py.runPython(
              'import sys\n' +
              'from js import prompt as _prompt\n' +
              'class _Stdin:\n' +
              '    def readline(self):\n' +
              '        r = _prompt("Input:")\n' +
              '        return ("" if r is None else r) + "\\n"\n' +
              'sys.stdin = _Stdin()'
            );
            resolve(py);
          }).catch(reject);
        };
        s.onerror = reject;
        document.head.appendChild(s);
      });
    }
    return _pyReady;
  }

  document.addEventListener('DOMContentLoaded', function () {
    /* Patch Python runner */
    var pyContainer = document.getElementById('runner-music-py');
    if (pyContainer) {
      var pyBtn = pyContainer.querySelector('.runBtn');
      if (pyBtn) {
        var newPyBtn = pyBtn.cloneNode(true);
        pyBtn.parentNode.replaceChild(newPyBtn, pyBtn);
        newPyBtn.addEventListener('click', async function () {
          var cm = pyContainer.querySelector('.CodeMirror') && pyContainer.querySelector('.CodeMirror').CodeMirror;
          if (!cm) return;
          var outEl = pyContainer.querySelector('.output-content');
          var etEl  = pyContainer.querySelector('.execTime');
          outEl.textContent = '⏳ Loading Python runtime…';
          if (etEl) etEl.textContent = '';
          var t0 = Date.now();
          try {
            var py = await getPyodide();
            var lines = [];
            py.setStdout({ batched: function (s) { lines.push(s); } });
            outEl.textContent = '⏳ Running…';
            await py.runPythonAsync(cm.getValue());
            outEl.textContent = lines.join('\n') || '[no output]';
            if (etEl) etEl.textContent = '⏱ Execution time: ' + (Date.now() - t0) + 'ms';
          } catch (e) {
            outEl.textContent = 'Error: ' + e.message;
            if (etEl) etEl.textContent = '';
          }
        });
      }
    }

    /* ── JavaScript: browser eval ── */
    var jsContainer = document.getElementById('runner-music-js');
    if (jsContainer) {
      var jsBtn = jsContainer.querySelector('.runBtn');
      if (jsBtn) {
        var newJsBtn = jsBtn.cloneNode(true);
        jsBtn.parentNode.replaceChild(newJsBtn, jsBtn);
        newJsBtn.addEventListener('click', function () {
          var cm = jsContainer.querySelector('.CodeMirror') && jsContainer.querySelector('.CodeMirror').CodeMirror;
          if (!cm) return;
          var outEl = jsContainer.querySelector('.output-content');
          var etEl  = jsContainer.querySelector('.execTime');
          outEl.textContent = '⏳ Running…';
          if (etEl) etEl.textContent = '';
          var t0 = Date.now(), logs = [], origLog = console.log;
          console.log = function () { logs.push(Array.from(arguments).map(String).join(' ')); origLog.apply(console, arguments); };
          try {
            eval(cm.getValue());
            console.log = origLog;
            outEl.textContent = logs.length ? logs.join('\n') : '[no output]';
            if (etEl) etEl.textContent = '⏱ Execution time: ' + (Date.now() - t0) + 'ms';
          } catch (e) {
            console.log = origLog;
            outEl.textContent = 'Error: ' + e.message;
            if (etEl) etEl.textContent = '';
          }
        });
      }
    }
  });
})();
